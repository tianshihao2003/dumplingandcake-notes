first_price = 0.4883  # 第一档单价

    # 第一档部分

    # 第二档部分
    second_usage = usage_elec - first_max

    total_cost = first_cost + second_tier_cost

# 需求17：【选做】模拟银行取款机
#         初始余额 10000 元，循环显示菜单：
#         1.查询余额  2.存款  3.取款  4.退出
#         取款金额不能超过余额，存款需为正数，退出时输出"感谢使用"
#         输入非菜单选项时提示"请重新选择"
# =====================================================================
print("\n" + "=" * 50)
print("【需求19】模拟银行取款机")

balance = 10000.0  # 初始余额

menu = """
# # # # # # # # # # # # # # # # # # # # # # # #
#      欢迎使用银行自助服务系统                   #
#      1. 查询余额                            #
#      2. 存款                               #
#      3. 取款                               #
#      4. 退出                               #
# # # # # # # # # # # # # # # # # # # # # # # #
"""

while True:
    print(menu)
    choice = input("请选择操作(1-4): ")#

    match choice:
        case "1":  # 查询余额
            print(f"当前余额: {balance:.2f}元")

        case "2":  # 存款
            amount = float(input("请输入存款金额: "))
            if amount > 0:
                balance += amount
                print(f"存款成功！当前余额: {balance:.2f}元")
            else:
                print("存款金额必须为正数！")

        case "3":  # 取款
            amount = float(input("请输入取款金额: "))
            if amount <= 0:
                print("取款金额必须为正数！")
            elif amount > balance:
                print(f"余额不足！当前余额: {balance:.2f}元")
            else:
                balance -= amount
                print(f"取款成功！当前余额: {balance:.2f}元")

        case "4":  # 退出
            print("感谢使用，再见 ~")
            break

        case _:
            print("输入有误，请重新选择！")
